# Internationalization (i18n)

> Localization setup, adding languages, testing.

## Overview

GitNotēs uses **i18next** for internationalization with **6 languages**: English, Spanish, French, German, Japanese, Korean.

## Setup

### Dependencies

```json
{
  "i18next": "^26.2.0",
  "react-i18next": "^17.0.8",
  "expo-localization": "~56.0.2"
}
```

### Configuration

```typescript
// src/i18n/index.ts

import i18n from 'i18next';
import { initReactI18next } from 'react-i18next';
import * as Localization from 'expo-localization';

import en from './en.json';
import es from './es.json';
import fr from './fr.json';
import de from './de.json';
import ja from './ja.json';
import ko from './ko.json';

const resources = {
  en: { translation: en },
  es: { translation: es },
  fr: { translation: fr },
  de: { translation: de },
  ja: { translation: ja },
  ko: { translation: ko },
};

// Detect device language
const getDeviceLanguage = () => {
  const locales = Localization.getLocales();
  const lang = locales[0]?.languageCode ?? 'en';
  
  // Map to supported languages
  const supported = ['en', 'es', 'fr', 'de', 'ja', 'ko'];
  return supported.includes(lang) ? lang : 'en';
};

i18n.use(initReactI18next).init({
  resources,
  lng: getDeviceLanguage(),
  fallbackLng: 'en',
  interpolation: {
    escapeValue: false,
  },
});

export default i18n;
```

## Usage

### In Components

```typescript
import { useTranslation } from 'react-i18next';

function SettingsScreen() {
  const { t } = useTranslation();
  
  return (
    <View>
      <Text>{t('settings.title')}</Text>
      <Text>{t('settings.dailyQuote.title')}</Text>
      <Text>{t('settings.aiPersonalization.title')}</Text>
    </View>
  );
}
```

### With Interpolation

```typescript
// en.json
{
  "greeting": "Hello, {{name}}!",
  "items": "{{count}} item",
  "items_plural": "{{count}} items"
}

// Component
<Text>{t('greeting', { name: 'Alice' })}</Text>
<Text>{t('items', { count: 5 })}</Text>
```

### Changing Language

```typescript
import { useTranslation } from 'react-i18next';

function LanguageSelector() {
  const { i18n } = useTranslation();
  
  const changeLanguage = (lang: string) => {
    i18n.changeLanguage(lang);
    // Persist to AsyncStorage
    StorageService.set('language', lang);
  };
  
  return (
    <View>
      {['en', 'es', 'fr', 'de', 'ja', 'ko'].map(lang => (
        <Button
          key={lang}
          title={lang}
          onPress={() => changeLanguage(lang)}
        />
      ))}
    </View>
  );
}
```

## Translation Guidelines

### Key Naming

Use **dot notation** for nested keys:

```json
{
  "settings": {
    "ai": {
      "personalization": {
        "title": "Personalize with my notes",
        "description": "When off, AI won't read your notes"
      }
    }
  }
}
```

### Placeholder Patterns

```json
{
  "welcome": "Welcome, {{name}}!",
  "items_count": "{{count}} item",
  "items_count_plural": "{{count}} items",
  "last_updated": "Updated {{date}}"
}
```

### Context-Aware Translations

```json
{
  "note": {
    "title": "Note",
    "title_plural": "Notes",
    "context": {
      "create": "Create note",
      "edit": "Edit note",
      "delete": "Delete note"
    }
  }
}
```

## Testing

### Unit Tests

```typescript
// __tests__/i18n.test.ts

import i18n from '../src/i18n';

describe('i18n', () => {
  it('has all languages loaded', () => {
    const languages = ['en', 'es', 'fr', 'de', 'ja', 'ko'];
    languages.forEach(lang => {
      expect(i18n.hasResourceBundle(lang, 'translation')).toBe(true);
    });
  });

  it('translates settings.dailyQuote.title', () => {
    i18n.changeLanguage('en');
    expect(i18n.t('settings.dailyQuote.title')).toBe('Daily Quote');
    
    i18n.changeLanguage('es');
    expect(i18n.t('settings.dailyQuote.title')).toBe('Cita Diaria');
  });

  it('falls back to English for missing keys', () => {
    i18n.changeLanguage('es');
    // Assume 'missing.key' only exists in 'en'
    expect(i18n.t('missing.key')).toBe('missing.key'); // Returns key
  });
});
```

### Component Tests

```typescript
// __tests__/components/SettingsScreen.test.tsx

import { I18nextProvider } from 'react-i18next';
import i18n from '../src/i18n';

describe('SettingsScreen', () => {
  it('renders in English', () => {
    i18n.changeLanguage('en');
    const { getByText } = render(
      <I18nextProvider i18n={i18n}>
        <SettingsScreen />
      </I18nextProvider>
    );
    expect(getByText('Settings')).toBeTruthy();
  });

  it('renders in Spanish', () => {
    i18n.changeLanguage('es');
    const { getByText } = render(
      <I18nextProvider i18n={i18n}>
        <SettingsScreen />
      </I18nextProvider>
    );
    expect(getByText('Configuración')).toBeTruthy();
  });
});
```

### Check Missing Keys

```typescript
// scripts/check-i18n.ts
import en from '../src/i18n/en.json';
import es from '../src/i18n/es.json';

function getKeys(obj: any, prefix = ''): string[] {
  return Object.keys(obj).flatMap(key => {
    const fullKey = prefix ? `${prefix}.${key}` : key;
    if (typeof obj[key] === 'object') {
      return getKeys(obj[key], fullKey);
    }
    return fullKey;
  });
}

const enKeys = getKeys(en);
const esKeys = getKeys(es);

const missing = enKeys.filter(key => !esKeys.includes(key));
console.log('Missing in Spanish:', missing);
```
